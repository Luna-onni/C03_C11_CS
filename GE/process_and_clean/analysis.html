prepare the data and check fixed effects

# transfer to factor and set the baseline
highest_cloze_all <- highest_cloze_all %>%
  ungroup() %>%
  mutate(
    participant = relevel(as.factor(participant), ref = "human"),
    Itemtype = relevel(as.factor(Itemtype), ref = "English"))

# check whether there are fixed effects
# for comparison, we need to set REML = FALSE because different fixed effect can not be compared with TRUE

# null model
m0 <- lmer(cloze_probability ~ 1 + (1 | inner_number), data = highest_cloze_all, REML = FALSE)

# participant
m1 <- lmer(cloze_probability ~ participant + (1 | inner_number), data = highest_cloze_all, REML = FALSE)
anova(m0, m1) # check whether participant has effect -> Yes, there is a participant effect

Participant effect: The highest cloze probability of human under English condition is 0.74, all LLMs have higher cloze probability than human (Claude = 0.26, GPT = 0.22, Mistral = 0.24), indicating that participant has significant effects and LLMs achieve higher cloze probability than human under English condition.

Itemtype effect: Humans are not sensitive to Itemtype (-0.009740).

Interaction effect: compare to human, Claude and Mistral are not affected by the Itemtype, while GPT achieves lower cloze probability under code-switching condition (GPT + English = 0.741 + 0.22 = 0.963, GPT + code-switching = 0.963 - 0.0097 - 0.0665 ≈ 0.887).
